        const filesContainer = document.getElementById('files');
        let currentFilter = 'all';

        // 2. 리스트 렌더링 함수
        function renderFiles() {
            filesContainer.innerHTML = '';
            
            const searchTerm = document.getElementById('search').value.toLowerCase();

            archiveData.forEach(item => {
                // 카테고리 필터링
                if (currentFilter !== 'all' && item.category !== currentFilter) return;
                
                // 검색어 필터링 (제목 검색)
                const titleLower = item.title.toLowerCase();
                const isMatch = searchTerm.length > 0 && titleLower.includes(searchTerm);
                const isHidden = searchTerm.length > 0 && !isMatch;

                if (isHidden) return; // 검색에 맞지 않으면 숨김

                const li = document.createElement('li');
                // 검색어가 있고 매칭되면 강조 표시 (요청하신 highlight 유지)
                if (isMatch) li.classList.add('highlight');

                li.innerHTML = `
                    <div class="thumb" style="background-color: ${item.color};"></div>
                    <div class="info">
                        <div class="title">${item.title}</div>
                        <span class="category">${item.category}</span>
                        <span class="date">${item.date}</span>
                    </div>
                `;
                
                // 클릭 시 모달 열기 이벤트 바인딩
                li.onclick = () => openModal(item);
                filesContainer.appendChild(li);
            });
        }

        // 3. 필터 버튼 이벤트
        document.querySelectorAll('.filter-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                // 버튼 활성화 스타일 변경
                document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
                e.target.classList.add('active');
                
                // 필터 값 변경 후 다시 렌더링
                currentFilter = e.target.getAttribute('data-filter');
                renderFiles();
            });
        });

        // 4. 검색창 이벤트
        document.getElementById('search').addEventListener('input', renderFiles);

        // 5. 모달 제어 함수
        function openModal(item) {
            document.getElementById('m-title').innerText = item.title;
            document.getElementById('m-category').innerText = item.category;
            document.getElementById('m-date').innerText = item.date;
            document.getElementById('m-desc').innerText = item.desc;
            document.getElementById('m-img').style.backgroundColor = item.color;
            document.getElementById('m-img').innerText = item.title + " (Image Area)";
            
            document.getElementById('modal').style.display = 'flex';
            document.body.style.overflow = 'hidden'; // 배경 스크롤 방지
        }

        function closeModal() {
            document.getElementById('modal').style.display = 'none';
            document.body.style.overflow = '';
